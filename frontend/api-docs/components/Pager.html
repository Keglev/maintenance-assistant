Description

The paging control the three tables share: a labelled arrow either side of "page X of Y".

The buttons carry their text, not only a chevron. The primary device is a shop-floor tablet operated with work gloves, and a bare arrow is both a smaller target and a guess — "Zurück" next to a table could as easily mean leaving the view. Spelling out "Vorherige Seite" / "Nächste Seite" costs width the page has and removes the guess.

Extracted rather than copied into each table because there are three of them now (the corpus, the archive and "Meine Uploads") and the disabled-at-the-ends rule is the kind of thing that gets fixed in one of three places.

Zero-based page in, one-based on screen: the API pages from 0 and people count from 1, and the conversion belongs in one place rather than at every call site.

import { Component, computed, inject, input, output } from '@angular/core';

import { I18nService } from '../../core/i18n/i18n.service';

/**
 * The paging control the three tables share: a labelled arrow either side of "page X of Y".
 *
 * <p><b>The buttons carry their text, not only a chevron.</b> The primary device is a shop-floor
 * tablet operated with work gloves, and a bare arrow is both a smaller target and a guess — "Zurück"
 * next to a table could as easily mean leaving the view. Spelling out "Vorherige Seite" / "Nächste
 * Seite" costs width the page has and removes the guess.
 *
 * <p>Extracted rather than copied into each table because there are three of them now (the corpus,
 * the archive and "Meine Uploads") and the disabled-at-the-ends rule is the kind of thing that gets
 * fixed in one of three places.
 *
 * <p>Zero-based {@link page} in, one-based on screen: the API pages from 0 and people count from 1,
 * and the conversion belongs in one place rather than at every call site.
 */
@Component({
  selector: 'app-pager',
  templateUrl: './pager.html',
  styleUrl: './pager.css',
})
export class Pager {
  private readonly i18n = inject(I18nService);

  protected readonly t = this.i18n.t;

  /** Zero-based, as the API counts. */
  readonly page = input.required<number>();
  readonly pageCount = input.required<number>();
  /** Rows in the whole result, so the control can say what it is paging through. */
  readonly total = input.required<number>();
  /** What the pager is called for a screen reader, when a view has more than one. */
  readonly label = input('');
  /** Suffix for the test ids, so two pagers on one screen are addressable apart. */
  readonly testId = input('pager');

  readonly previous = output<void>();
  readonly next = output<void>();

  protected readonly hasPrevious = computed(() => this.page() > 0);
  protected readonly hasNext = computed(() => this.page() + 1 < this.pageCount());
}

<nav class="pager" [attr.aria-label]="label() || t().pager.label">
  <button
    type="button"
    class="btn btn-secondary arrow-button"
    [attr.data-testid]="testId() + '-previous'"
    [disabled]="!hasPrevious()"
    (click)="previous.emit()"
  >
    <!-- Inline SVG like every other icon here: no icon CDN, because a third-party origin would be
         a DSGVO question and would be refused by the CSP. -->
    <svg class="arrow" viewBox="0 0 24 24" aria-hidden="true" focusable="false">
      <path d="M15 5l-7 7 7 7" />
    </svg>
    <span>{{ t().pager.previous }}</span>
  </button>

  <!-- aria-live so a screen reader hears which page it landed on; the buttons themselves keep
       focus, so nothing announces the whole table again. -->
  <span
    class="pager-state num"
    role="status"
    aria-live="polite"
    [attr.data-testid]="testId() + '-state'"
    >{{ t().pager.pageOf(page() + 1, pageCount(), total()) }}</span
  >

  <button
    type="button"
    class="btn btn-secondary arrow-button"
    [attr.data-testid]="testId() + '-next'"
    [disabled]="!hasNext()"
    (click)="next.emit()"
  >
    <span>{{ t().pager.next }}</span>
    <svg class="arrow" viewBox="0 0 24 24" aria-hidden="true" focusable="false">
      <path d="M9 5l7 7-7 7" />
    </svg>
  </button>
</nav>

.pager-state {
  font-size: var(--t-sm);
  color: var(--c-ink-muted);
  /* Between the two buttons on a wide row and on its own line when they wrap, rather than pinned to
     one side where it reads as belonging to whichever button it touches. */
  order: 0;
  flex: 1 1 auto;
  text-align: center;
}

/* Deliberately larger than .btn-sm: this is the control a gloved hand hits most often in a table,
   and the 44 px tap target is the design system's floor rather than its target. */
.arrow-button {
  display: inline-flex;
  align-items: center;
  gap: var(--s-2);
  min-height: var(--tap);
  padding-inline: var(--s-4);
  font-weight: 600;
}

.arrow {
  width: 1.25rem;
  height: 1.25rem;
  flex: none;
  fill: none;
  stroke: currentColor;
  stroke-width: 2;
  stroke-linecap: round;
  stroke-linejoin: round;
}

/* At the ends the button stays visible rather than disappearing: a control that vanishes moves the
   one next to it under the finger already travelling towards it. */
.arrow-button:disabled {
  opacity: 0.45;
  cursor: not-allowed;
}

/* Phone width: the label is what makes the arrows unambiguous, so the buttons keep their text and
   share the row, and the state line drops below them. */
@media (max-width: 30rem) {
  .arrow-button {
    flex: 1 1 auto;
    justify-content: center;
    padding-inline: var(--s-2);
  }

  .pager-state {
    order: 1;
    flex-basis: 100%;
  }
}

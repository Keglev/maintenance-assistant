Description

"Wie suche ich?" — the panel in the empty right half of the search view.

It is the designed answer to the lazy-query problem, not decoration. Retrieval here is semantic only, by architecture (ADR-004): there is no keyword index to fall back on, so a one-word question is not a narrower search, it is a worse one — the embedding of "frei" is close to nothing in particular. A user who does not know that types one word, gets Mode B, and concludes the assistant is broken. Explaining the rule beside the field costs a paragraph; the alternative is building a keyword search that was deliberately not built.

Its own component because the styling is a third of the search view's stylesheet and none of it is about answers. That also keeps `search.css` inside the per-component budget, which the combined file had just exceeded.

Subordinate by construction: no card, no shadow, helper-grey text behind a left rule. Once an answer arrives it must not compete with it — the answer is what the reader came for.

import { Component, inject, signal } from '@angular/core';

import { I18nService } from '../../core/i18n/i18n.service';

/**
 * "Wie suche ich?" — the panel in the empty right half of the search view.
 *
 * <p><b>It is the designed answer to the lazy-query problem, not decoration.</b> Retrieval here is
 * semantic only, by architecture (ADR-004): there is no keyword index to fall back on, so a
 * one-word question is not a narrower search, it is a worse one — the embedding of "frei" is close
 * to nothing in particular. A user who does not know that types one word, gets Mode B, and
 * concludes the assistant is broken. Explaining the rule beside the field costs a paragraph; the
 * alternative is building a keyword search that was deliberately not built.
 *
 * <p>Its own component because the styling is a third of the search view's stylesheet and none of
 * it is about answers. That also keeps `search.css` inside the per-component budget, which the
 * combined file had just exceeded.
 *
 * <p><b>Subordinate by construction</b>: no card, no shadow, helper-grey text behind a left rule.
 * Once an answer arrives it must not compete with it — the answer is what the reader came for.
 */
@Component({
  selector: 'app-search-help',
  templateUrl: './search-help.html',
  styleUrl: './search-help.css',
})
export class SearchHelp {
  private readonly i18n = inject(I18nService);

  protected readonly t = this.i18n.t;

  /**
   * Whether the panel is open.
   *
   * Only narrow viewports read this: on a wide screen the panel sits beside the form and the
   * stylesheet keeps it open regardless, because collapsing something already next to the field it
   * explains costs a click and buys nothing. Closed by default, so a phone shows the form first.
   */
  protected readonly open = signal(false);
}

<aside class="search-help" data-testid="search-help">
  <!-- The toggle is the heading on a narrow screen and hidden on a wide one, where the panel is
       simply open: a control that collapses something already beside the form buys nothing there. -->
  <button
    type="button"
    class="search-help-toggle"
    data-testid="search-help-toggle"
    [attr.aria-expanded]="open()"
    aria-controls="search-help-body"
    (click)="open.set(!open())"
  >
    <svg class="chevron" viewBox="0 0 24 24" aria-hidden="true" focusable="false">
      <path d="M6 9l6 6 6-6" />
    </svg>
    {{ t().search.helpToggle }}
  </button>

  <div class="search-help-body" id="search-help-body" [class.open]="open()">
    <h2 class="search-help-heading">{{ t().search.helpHeading }}</h2>
    <ul class="search-help-points">
      <li>{{ t().search.helpMeaning }}</li>
      <li>{{ t().search.helpContext }}</li>
      <li>{{ t().search.helpSentences }}</li>
    </ul>

    <h3 class="search-help-heading">{{ t().search.helpExamplesHeading }}</h3>
    <ul class="search-help-examples" data-testid="search-help-examples">
      @for (example of t().search.helpExamples; track example.good) {
        <li>
          <!-- Good and weak side by side. The weak half is what people actually type first, and
               showing it next to its fix teaches the rule faster than the rule does. -->
          <p class="example good">
            <span class="example-tag">{{ t().search.helpGood }}</span>
            <q>{{ example.good }}</q>
          </p>
          <p class="example weak">
            <span class="example-tag">{{ t().search.helpWeak }}</span>
            <q>{{ example.weak }}</q>
          </p>
        </li>
      }
    </ul>
  </div>
</aside>

/* Subordinate by construction: no card background, no shadow, a left rule and helper-grey text. It
   has to be readable and it must not compete with the answer that appears below it. */
:host {
  display: block;
  flex: 1 1 22rem;
  max-width: 34rem;
}

/* The inside of the gutter. --s-4 after a 2 px rule left the panel's own text hugging the line, and
   widening the gutter outside it only made that more obvious — a rule with air on one side and none
   on the other reads as a border on the text rather than as a divider between two columns. --s-5
   matches the space the answer block keeps inside its own edge. Overridden to 0 in the narrow block
   below, where the rule is horizontal and there is no gutter to be inside of. */
.search-help {
  padding-left: var(--s-5);
  border-left: 2px solid var(--c-line);
  font-size: var(--t-sm);
  color: var(--c-ink-muted);
}

.search-help-toggle {
  display: none;
  align-items: center;
  gap: var(--s-2);
  width: 100%;
  min-height: var(--tap);
  padding: 0;
  border: 0;
  background: none;
  font: inherit;
  font-weight: 600;
  color: var(--c-ink);
  cursor: pointer;
}

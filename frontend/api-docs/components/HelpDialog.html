import { Component, inject, input, output } from '@angular/core';

import { I18nService } from '../../core/i18n/i18n.service';
import { Dialog } from '../dialog/dialog';

/**
 * The help dialog: what the two answer modes mean, what each role may do, and three questions that
 * actually retrieve something.
 *
 * All the modal behaviour — focus trap, Esc, backdrop, the close button, full screen on a small
 * viewport — belongs to {@link Dialog} and is shared with the protocol viewer and the sign-out
 * confirmation. What is left here is the content, which is the only part specific to help.
 */
@Component({
  selector: 'app-help-dialog',
  imports: [Dialog],
  templateUrl: './help-dialog.html',
  styleUrl: './help-dialog.css',
})
export class HelpDialog {
  private readonly i18n = inject(I18nService);

  protected readonly t = this.i18n.t;

  readonly open = input(false);
  readonly closed = output<void>();
}

<app-dialog
  [open]="open()"
  [title]="t().help.title"
  testId="help"
  (closed)="closed.emit()"
>
  <section class="help-section">
    <h3 class="help-heading">{{ t().help.modesHeading }}</h3>
    <!-- The two modes are explained in their own colours: the reader meets the same green and
         the same amber here that the answer will arrive in. -->
    <div class="mode-card mode-card-a" data-testid="help-mode-a">
      <p class="mode-card-title">{{ t().help.modeAHeading }}</p>
      <p class="mode-card-body">{{ t().help.modeABody }}</p>
    </div>
    <div class="mode-card mode-card-b" data-testid="help-mode-b">
      <p class="mode-card-title">{{ t().help.modeBHeading }}</p>
      <p class="mode-card-body">{{ t().help.modeBBody }}</p>
    </div>
  </section>

  <section class="help-section">
    <h3 class="help-heading">{{ t().help.rolesHeading }}</h3>
    <ul class="help-list">
      <li>{{ t().help.roleOperator }}</li>
      <li>{{ t().help.roleTechniker }}</li>
      <li>{{ t().help.roleSchichtleiter }}</li>
    </ul>
  </section>

  <section class="help-section">
    <h3 class="help-heading">{{ t().demo.examplesHeading }}</h3>
    <p class="caption">{{ t().demo.examplesIntro }}</p>
    <ul class="examples">
      @for (example of t().demo.examples; track example.question) {
        <li class="example" data-testid="help-example">
          <p class="example-machine">
            {{ t().demo.exampleMachine }}: <strong>{{ example.machine }}</strong>
          </p>
          <p class="example-question" lang="de">{{ example.question }}</p>
          <p class="example-shows">{{ example.shows }}</p>
        </li>
      }
    </ul>
  </section>
</app-dialog>
